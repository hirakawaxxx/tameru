# tameru Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Markdown ナレッジベースを単一 Cloudflare Worker（Hono + D1）で提供し、MCP でエージェントが upsert/browse/search、ブラウザ Viewer でリスト・グラフ・本文を閲覧できるようにする。

**Architecture:** 1 Worker が `/mcp`（Streamable HTTP MCP）、Viewer HTML、`/api/*` 読み取り JSON を担う。`owner` は Cloudflare Access JWT のメール（小文字正規化）で全クエリをスコープ。本番デプロイは GitHub `main` → Workers Builds。ローカルは `wrangler dev` + ローカル D1。

**Tech Stack:** pnpm, TypeScript, Hono (`hono` + `@hono/zod-validator` + `@hono/mcp`), Cloudflare Workers / D1, Vitest, `@modelcontextprotocol/sdk` + `@hono/mcp` の `StreamableHTTPTransport`（Workers 向け Streamable HTTP）, design-kit（隣接 `../design-kit`）

## Global Constraints

- D1 のみ（R2・添付・ブラウザアップロードなし）
- 本文上限 **200 KiB**、tags 最大 **32**、links 最大 **128**、グラフノード上限 **2000**
- 他人の文書・存在しない ID・権限外 → **404**（存在の有無を漏らさない）
- 未認証 → 401、免責未同意 → Viewer インタースティシャル / API は同意系以外 **403**、バリデーション **400**、本文超過 **413**
- 本番で `workers.dev` 無効（または Access 必須）、PR プレビューデプロイなし（v1）
- 自前 `/admin` メトリクス UI・`metrics_daily` テーブルなし（Cloudflare Analytics のみ）
- ブラウザからの文書作成・更新・削除 API なし（MCP のみ書き込み）
- 免責 `disclaimer_version` 定数: `2026-11-14-v1`（改訂時に上げて再同意）
- コミット: Conventional Commits（`feat:` / `fix:` / `test:` / `chore:` / `docs:` / `ci:`）
- 仕様正本: `docs/superpowers/specs/2026-11-14-tameru-design.md`

CREATE TABLE users (
  owner TEXT PRIMARY KEY NOT NULL,
  disclaimer_version TEXT NOT NULL DEFAULT '0',
  disclaimer_accepted_at TEXT,
  created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE docs (
  id TEXT PRIMARY KEY NOT NULL,
  owner TEXT NOT NULL,
  slug TEXT,
  title TEXT NOT NULL,
  description TEXT NOT NULL DEFAULT '',
  kind TEXT NOT NULL DEFAULT 'doc',
  tags_json TEXT NOT NULL DEFAULT '[]',
  body_md TEXT NOT NULL,
  hash TEXT NOT NULL,
  links_json TEXT NOT NULL DEFAULT '[]',
  created_at TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at TEXT NOT NULL DEFAULT (datetime('now')),
  deleted_at TEXT
);

CREATE INDEX idx_docs_owner_updated
  ON docs(owner, updated_at DESC)
  WHERE deleted_at IS NULL;

CREATE INDEX idx_docs_owner_kind
  ON docs(owner, kind)
  WHERE deleted_at IS NULL;

CREATE TABLE doc_tags (
  owner TEXT NOT NULL,
  tag TEXT NOT NULL COLLATE NOCASE,
  doc_id TEXT NOT NULL,
  PRIMARY KEY (owner, tag, doc_id),
  FOREIGN KEY (doc_id) REFERENCES docs(id)
);

CREATE VIRTUAL TABLE docs_fts USING fts5(
  doc_id UNINDEXED,
  owner UNINDEXED,
  title,
  description,
  tags,
  body,
  tokenize='trigram'
);

CREATE TRIGGER docs_ai AFTER INSERT ON docs WHEN NEW.deleted_at IS NULL BEGIN
  INSERT INTO docs_fts(doc_id, owner, title, description, tags, body)
  VALUES (
    NEW.id,
    NEW.owner,
    NEW.title,
    NEW.description,
  (SELECT group_concat(value, ' ') FROM json_each(NEW.tags_json)),
    NEW.body_md
  );
END;

CREATE TRIGGER docs_au AFTER UPDATE ON docs BEGIN
  DELETE FROM docs_fts WHERE doc_id = OLD.id;
  INSERT INTO docs_fts(doc_id, owner, title, description, tags, body)
  SELECT NEW.id, NEW.owner, NEW.title, NEW.description,
    (SELECT group_concat(value, ' ') FROM json_each(NEW.tags_json)),
    NEW.body_md
  WHERE NEW.deleted_at IS NULL;
END;

CREATE TRIGGER docs_ad AFTER DELETE ON docs BEGIN
  DELETE FROM docs_fts WHERE doc_id = OLD.id;
END;
```

// src/lib/constants.ts
export const BODY_MAX_BYTES = 200 * 1024
export const MAX_TAGS = 32
export const MAX_LINKS = 128
export const GRAPH_MAX_NODES = 2000
export const SEARCH_MAX_TERMS = 8
export const BROWSE_MAX = 100

const ULID_RE = /[0-9A-HJKMNP-TV-Z]{26}/i
const LINK_RE = /\]\(\/d\/([0-9A-HJKMNP-TV-Z]{26})\)/gi

export function extractInternalLinks(markdown: string): string[] {
  const seen = new Set<string>()
  const out: string[] = []
  let m: RegExpExecArray | null
  const re = new RegExp(LINK_RE)
  while ((m = re.exec(markdown)) !== null) {
    const id = m[1].toUpperCase()
    if (!seen.has(id)) {
      seen.add(id)
      out.push(id)
    }
    if (out.length >= MAX_LINKS) break
  }
  return out
}
```

```typescript
// src/lib/markdown.ts
import matter from 'gray-matter'
import { extractInternalLinks, MAX_TAGS } from './constants'

export type ParsedDoc = {
  title: string
  description: string
  kind: string
  tags: string[]
  body: string
  links: string[]
}

function firstH1(body: string): string | null {
  const m = body.match(/^#\s+(.+)$/m)
  return m ? m[1].trim() : null
}

function normalizeTags(input: unknown): string[] {
  if (!Array.isArray(input)) return []
  return input
    .map((t) => String(t).trim())
    .filter(Boolean)
    .slice(0, MAX_TAGS)
}

export function parseMarkdownDoc(raw: string): ParsedDoc {
  const { data, content } = matter(raw)
  const body = content.trim()
  const title =
    (typeof data.title === 'string' && data.title.trim()) ||
    firstH1(body) ||
    '無題'
  const description =
    typeof data.description === 'string' ? data.description.trim() : ''
  const kind =
    typeof data.kind === 'string' && data.kind.trim() ? data.kind.trim() : 'doc'
  const tags = normalizeTags(data.tags)
  const links = extractInternalLinks(raw)
  return { title, description, kind, tags, body, links }
}

// src/lib/hash.ts
export async function sha256Hex(input: string): Promise<string> {
  const data = new TextEncoder().encode(input)
  const buf = await crypto.subtle.digest('SHA-256', data)
  return [...new Uint8Array(buf)].map((b) => b.toString(16).padStart(2, '0')).join('')
}

// src/lib/access.ts
import { createRemoteJWKSet, jwtVerify } from 'jose'

let jwks: ReturnType<typeof createRemoteJWKSet> | null = null

export function normalizeOwner(email: string): string {
  return email.trim().toLowerCase()
}

export function ownerFromDevHeaders(opts: {
  host: string
  accessAud: string
  debugOwner?: string | null
}): string | null {
  const isLocal =
    opts.host.startsWith('localhost') || opts.host.startsWith('127.0.0.1')
  if (!isLocal || opts.accessAud) return null
  if (!opts.debugOwner) return null
  return normalizeOwner(opts.debugOwner)
}

export async function verifyAccessJwt(
  token: string,
  teamDomain: string,
  aud: string,
): Promise<string> {
  const url = `https://${teamDomain}/cdn-cgi/access/certs`
  jwks ??= createRemoteJWKSet(new URL(url))
  const { payload } = await jwtVerify(token, jwks, {
    audience: aud,
    issuer: `https://${teamDomain}`,
  })
  const email = payload.email
  if (typeof email !== 'string' || !email) throw new Error('missing email')
  return normalizeOwner(email)
}
```

```typescript
// src/middleware/auth.ts
import { createMiddleware } from 'hono/factory'
import type { Env } from '../env'
import { ownerFromDevHeaders, verifyAccessJwt } from '../lib/access'

export type AuthVariables = { owner: string }

export const requireAuth = createMiddleware<{
  Bindings: Env
  Variables: AuthVariables
}>(async (c, next) => {
  const host = c.req.header('host') ?? ''
  const devOwner = ownerFromDevHeaders({
    host,
    accessAud: c.env.ACCESS_AUD,
    debugOwner: c.req.header('X-Tameru-Debug-Owner'),
  })
  if (devOwner) {
    c.set('owner', devOwner)
    return next()
  }
  if (!c.env.ACCESS_AUD) {
    return c.json({ error: 'unauthorized' }, 401)
  }
  const jwt = c.req.header('Cf-Access-Jwt-Assertion')
  if (!jwt) return c.json({ error: 'unauthorized' }, 401)
  try {
    const owner = await verifyAccessJwt(
      jwt,
      c.env.ACCESS_TEAM_DOMAIN,
      c.env.ACCESS_AUD,
    )
    c.set('owner', owner)
    await next()
  } catch {
    return c.json({ error: 'unauthorized' }, 401)
  }
})
```

// src/db/users.ts
export type UserRow = {
  owner: string
  disclaimer_version: string
  disclaimer_accepted_at: string | null
}

export function disclaimerAccepted(
  row: Pick<UserRow, 'disclaimer_version' | 'disclaimer_accepted_at'> | null,
  required: string,
): boolean {
  if (!row?.disclaimer_accepted_at) return false
  return row.disclaimer_version === required
}

export async function ensureUser(db: D1Database, owner: string): Promise<UserRow> {
  const existing = await db
    .prepare('SELECT owner, disclaimer_version, disclaimer_accepted_at FROM users WHERE owner = ?')
    .bind(owner)
    .first<UserRow>()
  if (existing) return existing
  await db
    .prepare('INSERT INTO users (owner) VALUES (?)')
    .bind(owner)
    .run()
  return {
    owner,
    disclaimer_version: '0',
    disclaimer_accepted_at: null,
  }
}

export async function acceptDisclaimer(
  db: D1Database,
  owner: string,
  version: string,
): Promise<void> {
  await db
    .prepare(
      `UPDATE users SET disclaimer_version = ?, disclaimer_accepted_at = datetime('now') WHERE owner = ?`,
    )
    .bind(version, owner)
    .run()
}

// 抜粋 src/db/docs.ts
export async function upsertDoc(
  db: D1Database,
  owner: string,
  args: { id?: string; markdown: string },
): Promise<
  | { status: 'created' | 'updated'; id: string }
  | { status: 'unchanged'; id: string }
  | { status: 'not_found' }
  | { status: 'too_large' }
> {
  const bytes = new TextEncoder().encode(args.markdown).byteLength
  if (bytes > BODY_MAX_BYTES) return { status: 'too_large' }
  const parsed = parseMarkdownDoc(args.markdown)
  const hash = await sha256Hex(args.markdown)
  const id = args.id ?? (await import('ulid')).ulid()
  if (args.id) {
    const row = await db
      .prepare('SELECT owner, hash, deleted_at FROM docs WHERE id = ?')
      .bind(id)
      .first<{ owner: string; hash: string; deleted_at: string | null }>()
    if (!row || row.owner !== owner || row.deleted_at)
      return { status: 'not_found' as const }
    if (row.hash === hash) return { status: 'unchanged', id }
  }
  const tagsJson = JSON.stringify(parsed.tags)
  const linksJson = JSON.stringify(parsed.links)
  const now = new Date().toISOString()
  await db.batch([
    db
      .prepare(
        `INSERT INTO docs (id, owner, title, description, kind, tags_json, body_md, hash, links_json, created_at, updated_at)
         VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
         ON CONFLICT(id) DO UPDATE SET
           title=excluded.title, description=excluded.description, kind=excluded.kind,
           tags_json=excluded.tags_json, body_md=excluded.body_md, hash=excluded.hash,
           links_json=excluded.links_json, updated_at=excluded.updated_at
         WHERE owner = excluded.owner AND deleted_at IS NULL`,
      )
      .bind(
        id,
        owner,
        parsed.title,
        parsed.description,
        parsed.kind,
        tagsJson,
        args.markdown,
        hash,
        linksJson,
        now,
        now,
      ),
    db.prepare('DELETE FROM doc_tags WHERE owner = ? AND doc_id = ?').bind(owner, id),
    ...parsed.tags.map((tag) =>
      db
        .prepare('INSERT INTO doc_tags (owner, tag, doc_id) VALUES (?, ?, ?)')
        .bind(owner, tag, id),
    ),
  ])
  const created = !args.id
  return { status: created ? 'created' : 'updated', id }
}
```

// src/lib/search.ts
import { SEARCH_MAX_TERMS } from './constants'

export function buildSearchPlan(q: string): {
  mode: 'fts' | 'like'
  ftsQuery?: string
  likePattern?: string
} {
  const terms = q
    .trim()
    .split(/\s+/)
    .filter(Boolean)
    .slice(0, SEARCH_MAX_TERMS)
  if (terms.length === 0) {
    return { mode: 'like', likePattern: '%' }
  }
  if (terms.some((t) => t.length < 3)) {
    return { mode: 'like', likePattern: `%${terms.join('%')}%` }
  }
  const ftsQuery = terms.map((t) => `"${t.replaceAll('"', '')}"`).join(' AND ')
  return { mode: 'fts', ftsQuery }
}
```

export type BrowseTarget =
  | { kind: 'root' }
  | { kind: 'recent' }
  | { kind: 'kinds' }

  | { kind: 'tags' }
  | { kind: 'tag'; value: string }

export function resolveBrowsePath(raw: string): BrowseTarget {
  const p = raw.replace(/\/+$/, '') || '/'
  if (p === '/') return { kind: 'root' }
  if (p === '/recent') return { kind: 'recent' }
  if (p === '/kind') return { kind: 'kinds' }
  if (p.startsWith('/kind/')) return { kind: 'kind', value: decodeURIComponent(p.slice(6)) }
  if (p === '/tags') return { kind: 'tags' }
  if (p.startsWith('/tags/')) return { kind: 'tag', value: decodeURIComponent(p.slice(6)) }
  return { kind: 'root' }
}

import { GRAPH_MAX_NODES } from '../lib/constants'

export async function getOwnerGraph(db: D1Database, owner: string) {
  const { results } = await db
    .prepare(
      `SELECT id, title, kind, links_json FROM docs
       WHERE owner = ? AND deleted_at IS NULL ORDER BY updated_at DESC`,
    )
    .bind(owner)
    .all<{ id: string; title: string; kind: string; links_json: string }>()
  const truncated = results.length > GRAPH_MAX_NODES
  const slice = truncated ? results.slice(0, GRAPH_MAX_NODES) : results
  const nodes = slice.map((r) => ({ id: r.id, title: r.title, kind: r.kind }))
  const idSet = new Set(nodes.map((n) => n.id))
  const edges: { from: string; to: string }[] = []
  for (const row of slice) {
    const links: string[] = JSON.parse(row.links_json || '[]')
    for (const to of links) {
      if (idSet.has(to)) edges.push({ from: row.id, to })
    }
  }
  return { nodes, edges, truncated }
}
